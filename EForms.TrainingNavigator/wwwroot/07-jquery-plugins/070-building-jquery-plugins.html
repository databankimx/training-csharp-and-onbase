1. Basic Plugin Structure

// A jQuery plugin is just a function attached to $.fn, which every jQuery
// object inherits. "this" inside it refers to the matched set of elements.
$.fn.greenify = function () {
    this.css("color", "green");
};

$("#basicTarget a").greenify();

2. Supporting Chaining

// jQuery methods are chainable because they return "this". A plugin that
// doesn't return anything breaks the chain for whoever calls it next.
$.fn.greenify = function () {
    this.css("color", "green");
    return this;
};

$("#chainingTarget a").greenify().addClass("greenified");

3. Protecting the $ Alias

// If another library also uses "$" (Prototype, MooTools, etc.), it can
// conflict with jQuery's own $ alias. Wrapping the plugin in an
// Immediately-Invoked Function Expression (IIFE) that takes jQuery as a
// parameter named $ guarantees $ means jQuery inside this function,
// no matter what it means anywhere else on the page.
(function ($) {
    $.fn.greenify = function () {
        this.css("color", "green");
        return this;
    };
}(jQuery));

4. Private, Scoped Variables

// The same IIFE wrapper also gives the plugin a private scope to keep
// its own variables in, invisible to and unaffected by the rest of the page.
(function ($) {
    var shade = "#556b2f";

    $.fn.greenify = function () {
        this.css("color", shade);
        return this;
    };
}(jQuery));

$("#privateVarsTarget a").greenify();
// "shade" does not exist out here -- try typing it in the console after running this

5. Minimizing Footprint

// Bad: two separate plugin methods, doubling the surface area to document,
// maintain, and guard against naming collisions.
$.fn.openPopup = function () { /* ... */ };
$.fn.closePopup = function () { /* ... */ };

// Better: one plugin, an argument selects the behavior.
$.fn.popup = function (action) {
    if (action === "open") { /* ... */ }
    if (action === "close") { /* ... */ }
};

6. Using .each()

// A plugin often needs to do something to EVERY matched element, not just
// the first one. Returning this.each(...) keeps the chain intact while
// iterating over the whole selection.
$.fn.myNewPlugin = function () {
    return this.each(function () {
        $(this).css("font-style", "italic");
    });
};

$("#eachTarget a").myNewPlugin();

7. Accepting Options

// $.extend merges an options object over a set of defaults, so callers
// only need to specify what they want to override.
(function ($) {
    $.fn.greenify = function (options) {
        var settings = $.extend({
            color: "#556b2f",
            backgroundColor: "white"
        }, options);
        return this.css({
            color: settings.color,
            backgroundColor: settings.backgroundColor
        });
    };
}(jQuery));

$("#optionsTarget div").greenify({ color: "orange" });

8. Putting It Together

(function ($) {
    $.fn.showLinkLocation = function () {
        this.filter("a").each(function () {
            var link = $(this);
            link.append(" (" + link.attr("href") + ")");
        });
        return this;
    };
}(jQuery));

$("#togetherTarget a").showLinkLocation();

9. An Optimized Variant

// .append() accepts a function that returns the content to append, called
// once per matched element -- avoiding the manual .each() loop above while
// doing exactly the same thing.
(function ($) {
    $.fn.showLinkLocation = function () {
        this.filter("a").append(function () {
            return " (" + this.href + ")";
        });
        return this;
    };
}(jQuery));

$("#optimizedTarget a").showLinkLocation();
